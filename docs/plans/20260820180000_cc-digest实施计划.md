// extract-sessions.js —— cc-digest 会话提取脚本(零依赖)
'use strict';
const fs = require('fs');
const path = require('path');
const os = require('os');
const readline = require('readline');
const { spawnSync } = require('child_process');

// 公私目录映射(静态已知信息;未知前缀进 ? 区,用户裁定后写回此处)
const COMPANY_PREFIXES = ['d--xunfei-'];            // xkzy 生态(大小写归一后匹配)
const PERSONAL_PREFIXES = ['d--study-', 'c--users-'];
const USER_MSG_MAX_LEN = 300;   // 提问候选只看短消息(长的是需求描述)
const KEEP_TEXT_LEN = 120;      // 聚合产物中消息截断长度
const QUESTION_RE = /(为什么|怎么回事|怎么办|怎么|如何|什么是|是什么|什么意思|哪些|哪个|怎样|吗|呢|[?？])/;
const HELP_RE = /(没看懂|看不懂|不理解|不懂|啥意思|又报错|还是不行|为什么总是|每次都|又要|又是)/;

function classifyDir(dirName) {
  const n = dirName.toLowerCase();
  if (COMPANY_PREFIXES.some(p => n.startsWith(p))) return 'company';
  if (PERSONAL_PREFIXES.some(p => n.startsWith(p))) return 'personal';
  return 'unknown';
}

function parseArgs(argv) {
  const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
  const opt = { since: null, until: null, scope: 'all', probe: false,
    root: path.join(os.homedir(), '.claude', 'projects'),
    out: path.join(os.homedir(), '.claude', 'cc-digest') };
  for (let i = 0; i < argv.length; i += 2) {
    const k = argv[i], v = argv[i + 1];
    if (k === '--since') opt.since = v;
    else if (k === '--until') opt.until = v;
    else if (k === '--scope') opt.scope = v;
    else if (k === '--root') opt.root = v;
    else if (k === '--out') opt.out = v;
    else if (k === '--probe' && (v === undefined || v.startsWith('--'))) { opt.probe = true; i -= 1; } // 布尔开关无值
    else throw new Error(`未知参数或缺值: ${k} ${v ?? ''}`);
  }
  if (!opt.since || !DATE_RE.test(opt.since)) throw new Error('--since 必填,格式 YYYY-MM-DD');
  if (!opt.until) opt.until = opt.since;
  if (!DATE_RE.test(opt.until)) throw new Error('--until 格式 YYYY-MM-DD');
  if (!['all', 'company', 'personal'].includes(opt.scope)) throw new Error('--scope 只能是 all|company|personal');
  return opt;
}

module.exports = { classifyDir, parseArgs, QUESTION_RE, HELP_RE };

// 追加到 extract-sessions.js(module.exports 之前)
function dayStart(s) { return new Date(s + 'T00:00:00'); }            // 本地时区日界
function dayEnd(s) { return new Date(s + 'T23:59:59.999'); }

function selectCandidateFiles(root, sinceMs) {
  const files = [];
  for (const dir of fs.readdirSync(root)) {
    const dirPath = path.join(root, dir);
    let st; try { st = fs.statSync(dirPath); } catch { continue; }
    if (!st.isDirectory()) continue;
    for (const f of fs.readdirSync(dirPath)) {
      if (!f.endsWith('.jsonl')) continue;
      const fp = path.join(dirPath, f);
      let fst; try { fst = fs.statSync(fp); } catch { continue; }
      if (fst.mtimeMs >= sinceMs) files.push(fp);   // 声音的预过滤:含窗口内时间戳的文件 mtime 必 ≥ 窗口起点
    }
  }
  return files;
}

function extractUserText(line) {
  if (line.type !== 'user' || line.isSidechain || line.isMeta) return null;
  const c = line.message && line.message.content;
  let text = '';
  if (typeof c === 'string') text = c;
  else if (Array.isArray(c)) text = c.filter(b => b.type === 'text').map(b => b.text).join('\n');
  text = (text || '').trim();
  if (!text) return null;
  if (text.startsWith('<command-name>') || text.startsWith('<local-command-stdout') || text.startsWith('<system-reminder>')) return null;
  return text;
}

function matchQuestion(text) {
  if (text.length > USER_MSG_MAX_LEN) return null;
  const q = text.match(QUESTION_RE); if (q) return q[1];
  const h = text.match(HELP_RE); if (h) return h[1];
  return null;
}

// 追加到 extract-sessions.js
function truncate(t, n) { return t.length > n ? t.slice(0, n) + '…' : t; }

function normalizeKey(text) {
  return text.toLowerCase().replace(/[0-9\s\p{P}\p{S}]+/gu, '').slice(0, 20);
}

async function aggregateSession(fp, sinceMs, untilMs) {
  const dir = path.basename(path.dirname(fp));
  const sess = { file: fp, dir, classify: classifyDir(dir), cwd: null, gitBranch: null,
    aiTitle: null, firstTs: null, lastTs: null, userCount: 0, assistantCount: 0,
    toolCount: 0, badLines: 0, userMsgs: [], questions: [], complaints: [] };
  const rl = readline.createInterface({ input: fs.createReadStream(fp, 'utf8') });
  for await (const raw of rl) {
    let o; try { o = JSON.parse(raw); } catch { sess.badLines++; continue; }
    if (o.cwd && !sess.cwd) sess.cwd = o.cwd;
    if (o.gitBranch && !sess.gitBranch) sess.gitBranch = o.gitBranch;
    if (o.aiTitle && !sess.aiTitle) sess.aiTitle = o.aiTitle;
    if (!o.timestamp) continue;
    const ts = new Date(o.timestamp).getTime();
    if (ts < sinceMs || ts > untilMs) continue;
    if (!sess.firstTs) sess.firstTs = o.timestamp;
    sess.lastTs = o.timestamp;
    const ut = extractUserText(o);
    if (ut) {
      sess.userCount++;
      sess.userMsgs.push({ ts: o.timestamp, text: truncate(ut, KEEP_TEXT_LEN) });
      const qf = matchQuestion(ut);
      if (qf) sess.questions.push({ ts: o.timestamp, text: truncate(ut, 100), feature: qf });
      const cf = ut.match(HELP_RE);
      if (cf) sess.complaints.push({ ts: o.timestamp, text: truncate(ut, 100), feature: cf[1] });
    }
    if (o.type === 'assistant' && !o.isSidechain) {
      sess.assistantCount++;
      const blocks = (o.message && o.message.content) || [];
      if (Array.isArray(blocks)) sess.toolCount += blocks.filter(b => b.type === 'tool_use').length;
    }
  }
  return sess;
}

function buildRepeatCandidates(sessions) {
  const map = new Map();
  for (const s of sessions) for (const m of s.userMsgs) {
    const key = normalizeKey(m.text);
    if (key.length < 6) continue;                       // 过短无语义
    if (!map.has(key)) map.set(key, { key, count: 0, samples: [] });
    const e = map.get(key);
    e.count++;
    if (e.samples.length < 3) e.samples.push({ dir: s.dir, ts: m.ts, text: m.text });
  }
  return [...map.values()].filter(e => e.count >= 2).sort((a, b) => b.count - a.count);
}

function buildMd(data) {
  const L = [];
  L.push(`# 会话活动清单 ${data.since} ~ ${data.until}`);
  L.push('');
  const byClass = { company: '## 🏢 公司(xkzy 生态)', personal: '## 🏠 个人', unknown: '## ❓ 未知目录(待裁定,请用户确认后写回脚本映射表)' };
  for (const [cls, title] of Object.entries(byClass)) {
    const group = data.sessions.filter(s => s.classify === cls);
    if (!group.length) continue;
    L.push(title, '');
    for (const s of group) {
      L.push(`### [${s.aiTitle || '(无标题)'}] \`${s.dir}\``, '');
      L.push(`- 时间:${s.firstTs} ~ ${s.lastTs} | 分支:${s.gitBranch || '-'} | 用户消息 ${s.userCount} / 回复 ${s.assistantCount} / 工具调用 ${s.toolCount}(坏行 ${s.badLines})`);
      for (const m of s.userMsgs.slice(0, 15)) L.push(`  - ${m.ts.slice(11, 16)} ${m.text}`);
      if (s.userMsgs.length > 15) L.push(`  - …另有 ${s.userMsgs.length - 15} 条`);
      for (const q of s.questions) L.push(`  - ❓[${q.feature}] ${q.text}`);
      L.push('');
    }
  }
  if (data.repeatCandidates.length) {
    L.push('## 🔁 重复模式候选(skill 提取建议原料)', '');
    for (const r of data.repeatCandidates) {
      L.push(`- **${r.key}** ×${r.count}`);
      for (const smp of r.samples) L.push(`  - ${smp.ts.slice(0, 10)} \`${smp.dir}\` "${smp.text}"`);
    }
    L.push('');
  }
  if (Object.keys(data.git).length) {
    L.push('## 📜 git 提交', '');
    for (const [cwd, logs] of Object.entries(data.git)) {
      L.push(`- **${cwd}**`);
      for (const g of logs) L.push(`  - ${g}`);
    }
    L.push('');
  }
  if (data.badLinesTotal) L.push(`> ⚠️ 共跳过损坏行 ${data.badLinesTotal} 行`, '');
  return L.join('\n');
}

async function runExtract(opts) {
  const sinceMs = dayStart(opts.since).getTime();
  const untilMs = dayEnd(opts.until).getTime();
  const files = selectCandidateFiles(opts.root, sinceMs);
  const sessions = [];
  for (const fp of files) {
    const s = await aggregateSession(fp, sinceMs, untilMs);
    if (s.firstTs) sessions.push(s);                    // 窗口内无活动的不进产物
  }
  const scoped = opts.scope === 'all' ? sessions : sessions.filter(s => s.classify === opts.scope);
  const data = { since: opts.since, until: opts.until, sessions: scoped,
    repeatCandidates: buildRepeatCandidates(scoped), git: {},          // Task 4 填充
    badLinesTotal: sessions.reduce((n, s) => n + s.badLines, 0) };
  fs.mkdirSync(opts.out, { recursive: true });
  const base = path.join(opts.out, `aggregate-${opts.since}_${opts.until}`);
  fs.writeFileSync(base + '.json', JSON.stringify(data, null, 2), 'utf8');
  fs.writeFileSync(base + '.md', buildMd(data), 'utf8');
  if (!scoped.length) console.log('NO_SESSIONS');
  else console.log(`${base}.md (${scoped.length} 会话)`);
}

  dayStart, dayEnd, selectCandidateFiles, extractUserText, matchQuestion,

if (require.main === module) {

// 追加到 extract-sessions.js。spawnSync 参数数组形式,规避 Windows cmd 引号问题
function collectGit(sessions, since, until) {
  const result = {};
  for (const cwd of new Set(sessions.map(s => s.cwd).filter(Boolean))) {
    if (!fs.existsSync(path.join(cwd, '.git'))) continue;
    const r = spawnSync('git', ['-C', cwd, 'log',
      `--since=${since}T00:00:00`, `--until=${until}T23:59:59`,
      '--pretty=format:%h|%ad|%s', '--date=format:%Y-%m-%d %H:%M', '--no-merges'],
      { encoding: 'utf8' });
    if (r.status !== 0 || !r.stdout.trim()) continue;   // 失败/无提交均跳过,不中断
    result[cwd] = r.stdout.trim().split('\n');
  }
  return result;
}

function runProbe(opts) {
